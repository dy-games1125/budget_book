      const savedBudgets = localStorage.getItem('pastel_budget_budgets');
      if (savedTx) transactions = JSON.parse(savedTx);
      if (savedBudgets) budgets = JSON.parse(savedBudgets);
    }

    // 데이터 저장
    function saveData() {
      localStorage.setItem('pastel_budget_tx', JSON.stringify(transactions));
      localStorage.setItem('pastel_budget_budgets', JSON.stringify(budgets));
    }

    // 현재 년-월 Key 구하기
    function getMonthKey() {
      const year = currentDate.getFullYear();
      const month = String(currentDate.getMonth() + 1).padStart(2, '0');
      return `${year}-${month}`;
    }

    // 월 이동
    function changeMonth(diff) {
      currentDate.setMonth(currentDate.getMonth() + diff);
      renderApp();
    }

    // 전체 뷰 렌더링
    function renderApp() {
      const monthKey = getMonthKey();
      document.getElementById('currentMonthText').innerText = `${currentDate.getFullYear()}년 ${currentDate.getMonth() + 1}월`;

      // 예산 정보 (없으면 기본값 0원)
      const currentBudget = budgets[monthKey] || 0;

      // 해당 월 내역 필터링
      const monthTx = transactions.filter(tx => tx.date.startsWith(monthKey));

      let totalIncome = 0;
      let totalExpense = 0;

      monthTx.forEach(tx => {
        if (tx.type === 'income') totalIncome += Number(tx.amount);
        else if (tx.type === 'expense') totalExpense += Number(tx.amount);
      });

      // 요약 카드 업데이트
      document.getElementById('totalIncomeDisplay').innerText = `+${totalIncome.toLocaleString()}원`;
      document.getElementById('totalExpenseDisplay').innerText = `-${totalExpense.toLocaleString()}원`;

      const remainingBudget = currentBudget - totalExpense;
      const budgetDisplay = document.getElementById('remainingBudgetDisplay');
      
      if (currentBudget === 0) {
        document.getElementById('budgetTitle').innerText = '🎯 설정된 예산이 없습니다';
        budgetDisplay.innerText = '0원 (예산을 설정해주세요)';
        budgetDisplay.style.color = 'var(--text-muted)';
        document.getElementById('budgetProgressBar').style.width = '0%';
      } else {
        document.getElementById('budgetTitle').innerText = `🎯 목표 예산: ${currentBudget.toLocaleString()}원`;
        if (remainingBudget >= 0) {
          budgetDisplay.innerText = `${remainingBudget.toLocaleString()}원 남음`;
          budgetDisplay.style.color = 'var(--text-main)';
        } else {
          budgetDisplay.innerText = `${Math.abs(remainingBudget).toLocaleString()}원 초과!`;
          budgetDisplay.style.color = 'var(--expense-color)';
        }

        // 프로그래스바 계산
        const usagePercent = Math.min(Math.round((totalExpense / currentBudget) * 100), 100);
        const progressBar = document.getElementById('budgetProgressBar');
        progressBar.style.width = `${usagePercent}%`;
        if (usagePercent > 90) progressBar.style.background = 'var(--expense-color)';
        else progressBar.style.background = 'linear-gradient(90deg, var(--secondary), var(--primary))';
      }

      // 타임라인 일기 UI 업데이트
      renderTimeline(monthTx);

      // 통계 그래프 업데이트
      renderCharts(monthTx);
    }

    // 타임라인 (가계부 일기) 렌더링
    function renderTimeline(monthTx) {
      const container = document.getElementById('timelineContainer');
      container.innerHTML = '';

      if (monthTx.length === 0) {
        container.innerHTML = `
          <div class="empty-state">
            <div style="font-size: 2.5rem;">📝</div>
            <p>이 달에 작성된 기록이 없어요.</p>
            <span style="font-size:0.8rem;">아래 + 버튼을 눌러 첫 내역을 적어보세요!</span>
          </div>
        `;
        return;
      }

      // 날짜 내림차순 정렬
      const sortedTx = [...monthTx].sort((a, b) => new Date(b.date) - new Date(a.date));

      // 날짜별 그룹화
      const grouped = {};
      sortedTx.forEach(tx => {
        if (!grouped[tx.date]) grouped[tx.date] = [];
        grouped[tx.date].push(tx);
      });

      for (const date in grouped) {
        const groupEl = document.createElement('div');
        groupEl.className = 'date-group';

        const dayOfWeek = ['일','월','화','수','목','금','토'][new Date(date).getDay()];
        const formattedDate = `${date.split('-')[1]}월 ${date.split('-')[2]}일 (${dayOfWeek})`;

        groupEl.innerHTML = `<div class="date-header"><span>${formattedDate}</span></div>`;

        grouped[date].forEach(tx => {
          const catList = CATEGORIES[tx.type] || [];
          const catObj = catList.find(c => c.id === tx.category) || { icon: '✨', label: tx.category };

          const card = document.createElement('div');
          card.className = 'diary-card';
          card.innerHTML = `
            <div class="diary-left">
              <div class="cat-icon">${catObj.icon}</div>
              <div class="diary-info">
                <span class="diary-title">${tx.title}</span>
                <span class="diary-memo">${catObj.label}</span>
              </div>
            </div>
            <div class="diary-right">
              <span class="diary-amount ${tx.type === 'income' ? 'income-text' : 'expense-text'}">
                ${tx.type === 'income' ? '+' : '-'}${Number(tx.amount).toLocaleString()}원
              </span>
              <div class="action-btns">
                <button class="mini-btn" onclick="openTransactionModal('${tx.id}')">✏️</button>
                <button class="mini-btn" onclick="deleteTransaction('${tx.id}')">🗑️</button>
              </div>
            </div>
          `;
          groupEl.appendChild(card);
        });

        container.appendChild(groupEl);
      }
    }

    // 통계 그래프 렌더링 (Chart.js)
    function renderCharts(monthTx) {
      // 1. 월별 수입/지출 바 차트
      let totalInc = 0;
      let totalExp = 0;
      monthTx.forEach(tx => {
        if (tx.type === 'income') totalInc += Number(tx.amount);
        else totalExp += Number(tx.amount);
      });

      const ctxBar = document.getElementById('barChart').getContext('2d');
      if (barChartInstance) barChartInstance.destroy();

      barChartInstance = new Chart(ctxBar, {
        type: 'bar',
        data: {
          labels: ['총 수입', '총 지출'],
          datasets: [{
            data: [totalInc, totalExp],
            backgroundColor: ['#70c1b3', '#f25f5c'],
            borderRadius: 10
          }]
        },
        options: {
          responsive: true,
          plugins: { legend: { display: false } },
          scales: { y: { beginAtZero: true } }
        }
      });

      // 2. 카테고리별 도넛 차트 (지출 기준)
      const expTx = monthTx.filter(tx => tx.type === 'expense');
      const catTotals = {};

      expTx.forEach(tx => {
        const catObj = CATEGORIES.expense.find(c => c.id === tx.category) || { label: '기타' };
        catTotals[catObj.label] = (catTotals[catObj.label] || 0) + Number(tx.amount);
      });

      const labels = Object.keys(catTotals);
      const data = Object.values(catTotals);

      const ctxDoughnut = document.getElementById('doughnutChart').getContext('2d');
      if (doughnutChartInstance) doughnutChartInstance.destroy();

      doughnutChartInstance = new Chart(ctxDoughnut, {
        type: 'doughnut',
        data: {
          labels: labels.length > 0 ? labels : ['지출 없음'],
          datasets: [{
            data: data.length > 0 ? data : [1],
            backgroundColor: labels.length > 0 
              ? ['#ffa8b6', '#93e1d8', '#ffd166', '#a2d2ff', '#cdb4db', '#b5e2fa', '#e63946', '#f4a261']
              : ['#e0e0e0']
          }]
        },
        options: {
          responsive: true,
          plugins: { legend: { position: 'bottom' } }
        }
      });
    }

    // 탭 전환
    function switchTab(tabName, el) {
      document.querySelectorAll('.nav-item').forEach(btn => btn.classList.remove('active'));
      document.querySelectorAll('.view-section').forEach(sec => sec.classList.remove('active'));

      el.classList.add('active');
      if (tabName === 'timeline') {
        document.getElementById('viewTimeline').classList.add('active');
      } else if (tabName === 'stats') {
        document.getElementById('viewStats').classList.add('active');
      }
    }

    // 모달 제어
    function openModal(id) { document.getElementById(id).classList.add('active'); }
    function closeModal(id) { document.getElementById(id).classList.remove('active'); }

    // 내역 작성/수정 모달 오픈
    function openTransactionModal(editId = null) {
      const todayStr = currentDate.toISOString().split('T')[0];
      
      if (editId) {
        const tx = transactions.find(t => t.id === editId);
        if (!tx) return;
        document.getElementById('txModalTitle').innerText = '내역 수정';
        document.getElementById('editTxId').value = tx.id;
        document.getElementById('txDate').value = tx.date;
        document.getElementById('txAmount').value = tx.amount;
        document.getElementById('txTitle').value = tx.title;
        setTxType(tx.type);
        selectCategory(tx.category);
      } else {
        document.getElementById('txModalTitle').innerText = '새 내역 작성';
        document.getElementById('editTxId').value = '';
        document.getElementById('txDate').value = todayStr;
        document.getElementById('txAmount').value = '';
        document.getElementById('txTitle').value = '';
        setTxType('expense');
      }
      openModal('txModal');
    }

    // 타입 설정 (지출/수입)
    function setTxType(type) {
      currentType = type;
      const btnExp = document.getElementById('btnExpense');
      const btnInc = document.getElementById('btnIncome');

      if (type === 'expense') {
        btnExp.className = 'type-btn active expense';
        btnInc.className = 'type-btn';
      } else {
        btnInc.className = 'type-btn active income';
        btnExp.className = 'type-btn';
      }
      renderCategorySelector();
    }

    // 카테고리 셀렉터 생성
    function renderCategorySelector() {
      const container = document.getElementById('catSelector');
      container.innerHTML = '';
      const catList = CATEGORIES[currentType];
      selectedCategory = catList[0].id;

      catList.forEach(cat => {
        const btn = document.createElement('button');
        btn.type = 'button';
        btn.className = `cat-btn ${cat.id === selectedCategory ? 'selected' : ''}`;
        btn.onclick = () => selectCategory(cat.id);
        btn.innerHTML = `<span>${cat.icon}</span><span>${cat.label}</span>`;
        btn.dataset.id = cat.id;
        container.appendChild(btn);
      });
    }

    function selectCategory(catId) {
      selectedCategory = catId;
      document.querySelectorAll('.cat-btn').forEach(b => {
        if (b.dataset.id === catId) b.classList.add('selected');
        else b.classList.remove('selected');
      });
    }

    // 내역 저장
    function saveTransaction() {
      const editId = document.getElementById('editTxId').value;
      const date = document.getElementById('txDate').value;
      const amount = document.getElementById('txAmount').value;
      const title = document.getElementById('txTitle').value.trim();

      if (!date || !amount || Number(amount) <= 0 || !title) {
        alert('모든 항목을 올바르게 입력해주세요!');
        return;
      }

      if (editId) {
        // 수정
        const index = transactions.findIndex(t => t.id === editId);
        if (index !== -1) {
          transactions[index] = { id: editId, type: currentType, date, amount: Number(amount), title, category: selectedCategory };
        }
      } else {
        // 추가
        const newTx = {
          id: 'tx_' + Date.now(),
          type: currentType,
          date,
          amount: Number(amount),
          title,
          category: selectedCategory
        };
        transactions.push(newTx);
      }

      saveData();
      closeModal('txModal');
      renderApp();
    }

    // 내역 삭제
    function deleteTransaction(id) {
      if (confirm('이 기록을 정말 삭제할까요?')) {
        transactions = transactions.filter(t => t.id !== id);
        saveData();
        renderApp();
      }
    }

    // 예산 모달
    function openBudgetModal() {
      const monthKey = getMonthKey();
      document.getElementById('budgetInput').value = budgets[monthKey] || '';
      openModal('budgetModal');
    }

    function saveBudget() {
      const val = document.getElementById('budgetInput').value;
      const monthKey = getMonthKey();
      budgets[monthKey] = Number(val) || 0;
      saveData();
      closeModal('budgetModal');
      renderApp();
    }

    // 백업 모달
    function openBackupModal() {
      openModal('backupModal');
    }

    // 내보내기 (JSON)
    function exportData() {
      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify({ transactions, budgets }));
      const downloadAnchor = document.createElement('a');
      downloadAnchor.setAttribute("href", dataStr);
      downloadAnchor.setAttribute("download", `가계부_백업_${new Date().toISOString().split('T')[0]}.json`);
      document.body.appendChild(downloadAnchor);
      downloadAnchor.click();
      downloadAnchor.remove();
    }

    // 불러오기 (JSON)
    function importData(event) {
      const file = event.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          const imported = JSON.parse(e.target.result);
          if (imported.transactions && imported.budgets) {
            transactions = imported.transactions;
            budgets = imported.budgets;
            saveData();
            renderApp();
            alert('성공적으로 데이터를 복구했습니다!');
            closeModal('backupModal');
          } else {
            alert('올바르지 않은 백업 파일 형식입니다.');
          }
        } catch (err) {
          alert('파일을 읽는 중 오류가 발생했습니다.');
        }
      };
      reader.readAsText(file);
    }
